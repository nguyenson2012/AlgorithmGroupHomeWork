// Time complexity: O(n)
// Space complexity: O(n) 
// Explanation: Using hash map to store frequency

class Solution {
    public int majorityElement(int[] nums) {
        HashMap<Integer, Integer> hm = new HashMap<>();
        int res = 0;
        for(int num : nums){
            hm.put(num, hm.getOrDefault(num, 0) + 1);
            if(hm.get(num) > nums.length/2){
                res = num;
            }
        }
        return res;
    }
}
